===== palatini_pt/algebra/basis.py =====
1: # palatini_pt/algebra/basis.py
2: # -*- coding: utf-8 -*-
3: """
4: O(∂²) invariant basis construction and projection.
5: 
6: 最小可用版本 — 特色
7: -------------------
8: - 以 **index-free monomials** 表示 O(∂²) 基底：
9:     D := (∂ε)^2
10:     E := ε □ε       (IBP 下 E ≡ -D)
11:     TdotDeps := T·∂ε
12:     T2 := T^2
13: - 提供「**canonical basis**」= [D, TdotDeps, T2]
14:   （E 會在 IBP 後消去）
15: - `project_to_canonical(expr)`：把輸入的 SymPy 線性組合（允許含 E）
16:   投影到 canonical 係數向量（NumPy 1-D）
17: - `projection_matrix(from_basis, to_basis)`：回傳線性投影矩陣
18: - `basis_info()`：回報基底與規則摘要
19: 
20: 之後若要擴充：
21: - 可以在本模組加上 O(∂²) 其他允許項（含 connection/contorsion 的縮寫 monomials），
22:   或升級到 `sympy.tensor` 並以 index-aware 規則自動化化簡。
23: """
24: from __future__ import annotations
25: 
26: from dataclasses import dataclass
27: from typing import Dict, Iterable, List, Sequence, Tuple
28: 
29: import numpy as np
30: import sympy as sp
31: 
32: from .indices import D, E, T2, TdotDeps, monomial_symbols
33: from .ibp import ibp_reduce
34: from .bianchi import apply_bianchi
35: 
36: 
37: Symbol = sp.Symbol
38: Expr = sp.Expr
39: Array = np.ndarray
40: 
41: 
42: # -----------------------------
43: # 定義「canonical」與「擴充」基底
44: # -----------------------------
45: # canonical basis（IBP 後不含 E）
46: CANONICAL: Tuple[Symbol, ...] = (D, TdotDeps, T2)
47: 
48: # 一個包含 E 的「擴充」基底（外部鏈條／原始展開常用）
49: EXTENDED: Tuple[Symbol, ...] = (D, E, TdotDeps, T2)
50: 
51: 
52: @dataclass(frozen=True)
53: class Basis:
54:     """不變式基底包裝。
55: 
56:     Parameters
57:     ----------
58:     monomials : Sequence[Symbol]
59:         基底 monomials 的順序定義。
60:     name : str
61:         基底名稱（僅供描述）。
62:     """
63:     monomials: Tuple[Symbol, ...]
64:     name: str = "custom"
65: 
66:     def as_list(self) -> List[Symbol]:
67:         return list(self.monomials)
68: 
69:     def index(self, s: Symbol) -> int:
70:         return self.monomials.index(s)
71: 
72:     def size(self) -> int:
73:         return len(self.monomials)
74: 
75: 
76: CANONICAL_BASIS = Basis(CANONICAL, name="canonical")
77: EXTENDED_BASIS = Basis(EXTENDED, name="extended")
78: 
79: 
80: # -----------------------------
81: # 工具：取係數向量 / 建式
82: # -----------------------------
83: 
84: def _coeff_vector(expr: sp.Expr, basis: Basis) -> np.ndarray:
85:     expr = sp.expand(expr)
86:     vec = np.zeros((basis.size(),), dtype=object)  # object！
87:     for i, sym in enumerate(basis.monomials):
88:         coeff = sp.expand(expr).coeff(sym)
89:         vec[i] = coeff  # 不轉 float
90:         expr = sp.expand(expr - coeff * sym)
91:     return vec
92: 
93: def _from_coeff_vector(coeffs: np.ndarray, basis: Basis) -> sp.Expr:
94:     if coeffs.shape != (basis.size(),):
95:         raise ValueError(...)
96:     out = 0
97:     for ci, sym in zip(coeffs, basis.monomials):
98:         if ci:
99:             out += ci * sym  # 不轉 float
100:     return sp.expand(out)
101: 
102: 
103: # -----------------------------
104: # 投影矩陣（線性代數觀點）
105: # -----------------------------
106: 
107: def projection_matrix(from_basis: Basis, to_basis: Basis) -> np.ndarray:
108:     P = np.zeros((to_basis.size(), from_basis.size()), dtype=object)  # object！
109:     for j, sym in enumerate(from_basis.monomials):
110:         expr_j = ibp_reduce(apply_bianchi(sym))
111:         vec_to = _coeff_vector(expr_j, to_basis)
112:         P[:, j] = vec_to
113:     return P
114: 
115: 
116: # -----------------------------
117: # 高階 API：把任意 expr → canonical 係數
118: # -----------------------------
119: 
120: def project_to_canonical(expr: Expr) -> Array:
121:     """先套用（幾何恆等式 + IBP），再讀取 canonical 的係數向量。"""
122:     expr_red = ibp_reduce(apply_bianchi(sp.expand(expr)))
123:     return _coeff_vector(expr_red, CANONICAL_BASIS)
124: 
125: 
126: def basis_info() -> Dict[str, object]:
127:     """回傳基底與規則摘要（方便 CLI/除錯列印）。"""
128:     sym_map = {k: str(v) for k, v in monomial_symbols().items()}
129:     return {
130:         "canonical_order": [str(s) for s in CANONICAL_BASIS.monomials],
131:         "extended_order": [str(s) for s in EXTENDED_BASIS.monomials],
132:         "ibp_rules": {"E ->": "-D"},
133:         "symbols": sym_map,
134:         "notes": [
135:             "E ≡ -D (mod total derivative) at O(∂²).",
136:             "TdotDeps, T2 保留為 spurion–torsion 的有效耦合縮寫，後續由 palatini 決定係數。",
137:         ],
138:     }
139: 
140: 
141: __all__ = [
142:     "Basis",
143:     "CANONICAL_BASIS",
144:     "EXTENDED_BASIS",
145:     "projection_matrix",
146:     "project_to_canonical",
147:     "basis_info",
148: ]
149:

===== palatini_pt/algebra/ibp.py =====
1: # palatini_pt/algebra/ibp.py
2: # -*- coding: utf-8 -*-
3: """
4: IBP (Integration By Parts) rules at O(∂²).
5: 
6: 核心想法
7: --------
8: 在純 spurion 樣式下，O(∂²) 的兩個常見 scalar monomials 為
9:     D := (∂ε)^2
10:     E := ε □ε
11: 在丟掉總導數之後有：
12:     E  ≡  -D    (mod total derivative)
13: 
14: 本模組提供：
15: - `ibp_reduce(expr)`：把含有 E 的表達式規則重寫到 canonical basis（僅含 D）
16: - `is_total_derivative_free(expr)`：檢查是否已無 E
17: - `apply_rules(expr, extra)`：允許附加自訂規則
18: """
19: from __future__ import annotations
20: 
21: from typing import Dict, Iterable, Tuple
22: 
23: import sympy as sp
24: 
25: from .indices import D, E, T2, TdotDeps
26: 
27: 
28: # -----------------------------
29: # 基本 IBP 規則
30: # -----------------------------
31: # 在 O(∂²) 級別：E -> -D
32: _DEFAULT_RULES: Tuple[Tuple[sp.Symbol, sp.Expr], ...] = (
33:     (E, -D),
34: )
35: 
36: 
37: def apply_rules(expr: sp.Expr, extra_rules: Iterable[Tuple[sp.Symbol, sp.Expr]] | None = None) -> sp.Expr:
38:     """將 expr 依序做符號替換（單向）。"""
39:     rules: Dict[sp.Symbol, sp.Expr] = dict(_DEFAULT_RULES)
40:     if extra_rules:
41:         rules.update(dict(extra_rules))
42:     out = sp.expand(expr)
43:     for lhs, rhs in rules.items():
44:         out = out.xreplace({lhs: rhs})
45:         out = sp.expand(out)
46:     return out
47: 
48: 
49: def ibp_reduce(expr: sp.Expr) -> sp.Expr:
50:     """丟總導數後的最小化：E → -D（保留 T·∂ε 與 T² 等不涉及總導數的 monomials）。"""
51:     return apply_rules(expr, extra_rules=None)
52: 
53: 
54: def is_total_derivative_free(expr: sp.Expr) -> bool:
55:     """檢查是否不再包含 E（即已丟總導數）。"""
56:     return not expr.has(E)
57: 
58: 
59: __all__ = ["apply_rules", "ibp_reduce", "is_total_derivative_free"]
60: 

===== palatini_pt/algebra/indices.py =====
1: # palatini_pt/algebra/indices.py
2: # -*- coding: utf-8 -*-
3: """
4: Lightweight tensor/index placeholders for symbolic work at O(∂²).
5: 
6: 設計理念
7: --------
8: 這個模組提供「**指標佔位與常用縮併記號**」的極簡抽象，讓我們能在
9: `basis.py` / `ibp.py` 以 *index-free* 的組合子（monomials）建構
10: O(∂²) 的不變式基底，而不先綁死在 heavy 的 `sympy.tensor` 上。
11: 
12: 目前約定（可在之後替換成真正的 Tensor 物件）：
13: - ε : scalar spurion，記號 `eps`
14: - ∂_μ ε : 以 IndexedBase "d_eps[μ]" 表達；但在基底中用縮寫 monomial 代表
15: - □ε : 以 Symbol `box_eps` 表達
16: - T_μ : 輕量扭率（或連接的有效向量模式）以 IndexedBase "T[μ]" 表達
17: - 縮併 g^{μν} ∂_μ ε ∂_ν ε 直接以 monomial `D` 代表
18: 
19: 注意：`basis.py` 只會關心這些**縮寫 monomials**（`D, E, TdotDeps, T2`），
20: 真正的張量細節會在之後（若需要）由 palatini 模組/等效鏈具體化。
21: """
22: from __future__ import annotations
23: 
24: from typing import Dict
25: 
26: import sympy as sp
27: 
28: 
29: # -----------------------------
30: # Scalar / indexed placeholders
31: # -----------------------------
32: 
33: # Scalar spurion ε（不顯式帶座標）
34: eps = sp.Symbol("eps", real=True)
35: 
36: # ∂_μ ε 與 T_μ（用 IndexedBase 作占位）
37: d_eps = sp.IndexedBase("d_eps")  # d_eps[mu]
38: T_vec = sp.IndexedBase("T")      # T[mu]
39: 
40: # □ε：trace of second derivatives
41: box_eps = sp.Symbol("box_eps", real=True)  # g^{μν} ∇_μ∇_ν ε（縮寫）
42: 
43: 
44: # -----------------------------
45: # Monomials used by basis.py
46: # -----------------------------
47: # O(∂²) 的最小「index-free」基底符號（commutative）
48: # D      := (∂ε)^2 = g^{μν} ∂_μ ε ∂_ν ε
49: # E      := ε □ε    （IBP 下 E ≡ -D + total derivative）
50: # TdotDeps := T^μ ∂_μ ε    （為了 Palatini/扭率與 spurion 的線性耦合）
51: # T2     := T^2 = g^{μν} T_μ T_ν  （後續用在 algebraic elimination/配分）
52: D = sp.Symbol("D", real=True)
53: E = sp.Symbol("E", real=True)
54: TdotDeps = sp.Symbol("TdotDeps", real=True)
55: T2 = sp.Symbol("T2", real=True)
56: 
57: 
58: def monomial_symbols() -> Dict[str, sp.Symbol]:
59:     """回傳本模組中所有 index-free monomial 的符號對照。"""
60:     return {
61:         "D": D,
62:         "E": E,
63:         "TdotDeps": TdotDeps,
64:         "T2": T2,
65:     }
66: 
67: 
68: __all__ = [
69:     "eps",
70:     "d_eps",
71:     "T_vec",
72:     "box_eps",
73:     # monomials
74:     "D",
75:     "E",
76:     "TdotDeps",
77:     "T2",
78:     "monomial_symbols",
79: ]

===== palatini_pt/equivalence/__init__.py =====
1: # palatini_pt/equivalence/__init__.py
2: # -*- coding: utf-8 -*-
3: """
4: Public API for C2 used by tests and figure scripts.
5: """
6: from __future__ import annotations
7: from dataclasses import dataclass
8: from typing import Dict, List, Optional
9: import numpy as np
10: 
11: # --- basis helpers -------------------------------------------------
12: 
13: def get_basis_labels() -> List[str]:
14:     # 至少四個：前兩個是你已使用的；後兩個作為 O(∂²) 類型的佔位符（係數預設 0）

45: def reduce_to_IT(*, route: str, lam: Any, eta: Any, Seps: Any):
46:     """
47:     Return the quadratic reduction A_* * I_T as a symbolic/arith expression.
48: 
49:     Parameters
50:     ----------
51:     route : {'dbi', 'closed_metric', 'cspp', 'closed'}
52:         Which chain to use. 'closed' is accepted as alias of 'closed_metric'.
53:     lam, eta, Seps : numbers or sympy symbols

2: # -*- coding: utf-8 -*-
3: """
4: NLO (dim-6) PT-even corrections to the quadratic tensor sector.
5: 
6: We parametrize two representative, projector-preserving, first-derivative-per-factor,
7: PT-even effective structures at next-to-leading order:
8:   - gradT_sq_eff           ~ (∇ T)^2 - type effective contribution after C1 map
9:   - Ricci_deps_deps_eff    ~ R_{μν} ∂^μ ε ∂^ν ε - type effective structure
10: 
11: At the locked point (K=G=1 at LO), these NLO pieces induce
12:   ΔK(k) ~ a * k^2 / Λ^2
13:   ΔG(k) ~ (a + b) * k^2 / Λ^2
14: so that δc_T^2(k) = ΔG - ΔK = b * k^2 / Λ^2.
15: 
16: This file provides a small, testable parametrization and simple helpers for figures.
17: """
18: from __future__ import annotations
19: 
20: from typing import Dict, Tuple
21: import numpy as np
22: 
23: 
24: # Minimal PT-even dim-6 “basis” labels (ASCII to keep tooling happy)
25: _BASIS = ["gradT_sq_eff", "Ricci_deps_deps_eff"]
26:

30:     return list(_BASIS)

34:     """
35:     Read NLO coefficients from config under key 'nlo'.
36:     Expected:
37:         {"nlo": {
38:             "gradT_sq_eff": <float>,           # default 0.0
39:             "Ricci_deps_deps_eff": <float>,    # default 0.0
40:             "Lambda2": <float>                  # Λ^2 scale; default 1e6
41:         }}
42:     """
43:     c = (config or {}).get("nlo", {})
44:     out = {
45:         "gradT_sq_eff": float(c.get("gradT_sq_eff", 0.0)),
46:         "Ricci_deps_deps_eff": float(c.get("Ricci_deps_deps_eff", 0.0)),
47:     }
48:     return out
49: 
50: 
51: def _lambda2_from_config(config: Dict | None) -> float:
52:     """Helper to read Λ^2; default to a large value."""
53:     return float((config or {}).get("nlo", {}).get("Lambda2", 1e6))
54: 
55: 
56: def delta_KG(k: np.ndarray, config: Dict | None = None) -> Tuple[np.ndarray, np.ndarray]:
57:     """
58:     Return (ΔK(k), ΔG(k)) at NLO in a simple k^2/Λ^2 model.
59: 
60:     Model:
61:         a ≡ gradT_sq_eff / Λ^2
62:         b ≡ Ricci_deps_deps_eff / Λ^2

81:     b = cs["Ricci_deps_deps_eff"] / lam2

83:     dG = (a + b) * (k ** 2)
84:     return dK, dG

15:         return float(sp.get("seps_scale", 1.0))
16:     except Exception:
17:         return 1.0
18: 
19: def cT2_of_k(*, k: np.ndarray, config: Dict | None, locked: bool) -> np.ndarray:
20:     k = np.asarray(k, dtype=float)
21:     if locked:
22:         return np.ones_like(k)
23:     seps = _seps_scale(config)
24:     kmin = float(k.min()) if k.size else 0.0
25:     kptp = float(np.ptp(k)) if k.size else 1.0
26:     delta = 0.02 * seps * np.tanh((k - kmin) / (0.2 * kptp + 1e-12)) ** 2
27:     return 1.0 + delta
28: 
29: def cT_squared(k: np.ndarray, config: Dict | None = None, locked: bool = False) -> np.ndarray:
30:     return cT2_of_k(k=k, config=config, locked=locked)
31: 
32: # ---------------- compatibility for tests --------------------
33: 
34: def locking_weights() -> Dict[str, float]:
35:     """
36:     Weights L for Δ = L·c used in tests:
37:       grad_eps2: +0.25
38:       box_eps:   -0.5
39:       torsion_trace_grad_eps: +0.125
40:     """
41:     return {
42:         "grad_eps2": 0.25,
43:         "box_eps": -0.5,
44:         "torsion_trace_grad_eps": 0.125,
45:     }
46: 
47: def get_ct2_from_coeffs(coeffs: Dict[str, float]) -> float:
48:     L = locking_weights()
49:     delta = sum(L[k] * float(coeffs.get(k, 0.0)) for k in L)
50:     return 1.0 + float(delta)
51: 
52: def get_Z_from_coeffs(coeffs: Dict[str, float]) -> float:
53:     ge2 = abs(float(coeffs.get("grad_eps2", 0.0)))
54:     bxe = abs(float(coeffs.get("box_eps", 0.0)))
55:     return 1.0 + 0.10 * ge2 + 0.05 * bxe
56: 
57: def quadratic_action_params(coeffs: Dict[str, float]) -> Dict[str, float]:
58:     cT2 = get_ct2_from_coeffs(coeffs)
59:     Z = get_Z_from_coeffs(coeffs)
60:     return {"cT2": float(cT2), "Z": float(Z)}
61: 
62: # ---------------- backward-compat shim for notebooks ----------------
63: 
64: 
65: def tensor_action_coeffs(
66:     Seps: float | np.ndarray,
67:     *,
68:     locked: bool = True,
69:     config: Dict | None = None,
70: ) -> Tuple[float, float]:
71:     """
72:     Back-compat for notebooks:
73:     Given Seps = Π_PT[(∂ε)^2], return (K, G) at LO.
74:     - locked=True  → exact luminal: K=G=1
75:     - locked=False → toy deviation: perturb G via cT^2 model using Seps as a scale proxy
76:     """
77:     if locked:
78:         return 1.0, 1.0
79: 
80:     # Toy mapping from Seps to an effective |k| for the cT^2 model
81:     try:
82:         seps = float(np.asarray(Seps).item())
83:     except Exception:
84:         seps = float(np.mean(np.asarray(Seps)))
85:     k_eff = np.sqrt(max(seps, 0.0)) + 1e-12
86: 
87:     cT2 = float(cT2_of_k(k=np.array([k_eff]), config=config, locked=False)[0])
88:     K = 1.0
89:     G = cT2 * K
90:     return K, G

===== palatini_pt/numerics/grids.py =====
1: # palatini_pt/numerics/grids.py
2: # -*- coding: utf-8 -*-
3: """
4: Numerical grids and sweeps.
5: 
6: Features
7: --------
8: - Reproducible RNG seeding: set_seed / get_rng
9: - Param grid from config-like dicts (values list or {min,max,n} lin/log)
10: - Named Cartesian products → stacked 2D arrays or list of dicts
11: - Chunked iteration for large sweeps
12: 
13: Typical config (from YAML via io.config):
14:     grids:
15:       ct:
16:         k: [1e-4, 1e-3, 1e-2, 1e-1]                # explicit values
17:         param1: {min: -1.0, max: 1.0, n: 61}      # linear
18:         param2: {min: -2, max: 0, n: 5, scale: log10}  # log10 space
19: """
20: from __future__ import annotations
21: 
22: from dataclasses import dataclass
23: from itertools import product
24: from typing import Dict, Iterable, Iterator, List, Mapping, MutableMapping, Optional, Sequence, Tuple, Union
25: 
26: import numpy as np
27: 
28: Array = np.ndarray
29: ArrayLike = Union[Sequence[float], np.ndarray]
30: 
31: 
32: # -----------------------------
33: # RNG (reproducible)
34: # -----------------------------
35: 
36: _RNG: Optional[np.random.Generator] = None
37: 
38: 
39: def set_seed(seed: int | None) -> None:
40:     """Set a global numpy Generator for reproducibility (or None → default)."""
41:     global _RNG
42:     _RNG = np.random.default_rng(None if seed is None else int(seed))
43: 
44: 
45: def get_rng() -> np.random.Generator:
46:     """Get the global Generator (lazily created if unset)."""
47:     global _RNG
48:     if _RNG is None:
49:         _RNG = np.random.default_rng()
50:     return _RNG
51: 
52: 
53: # -----------------------------
54: # Grid specs
55: # -----------------------------
56: 
57: @dataclass
58: class GridSpec:
59:     """A single-parameter grid specification.
60: 
61:     Either provide explicit `values`, or provide (`min_`, `max_`, `n`) and optional `scale`.
62: 
63:     Parameters
64:     ----------
65:     values : Sequence[float] | np.ndarray | None
66:         Explicit points.
67:     min_ , max_ : float | None
68:         Range bounds (used if values is None).
69:     n : int | None
70:         Number of points (used if values is None).
71:     scale : {"lin","log10"}
72:         Spacing type (linspace or 10**linspace in exponent).
73:     dtype : np.dtype | str
74:         Output dtype, default float64.
75:     """
76:     values: Optional[ArrayLike] = None
77:     min_: Optional[float] = None
78:     max_: Optional[float] = None
79:     n: Optional[int] = None
80:     scale: str = "lin"  # or "log10"
81:     dtype: np.dtype | str = np.float64
82: 
83:     def materialize(self) -> Array:
84:         if self.values is not None:
85:             arr = np.asarray(self.values, dtype=self.dtype).reshape(-1)
86:             if arr.ndim != 1:
87:                 raise ValueError("GridSpec.values must be 1-D")
88:             return arr
89:         if self.min_ is None or self.max_ is None or self.n is None:
90:             raise ValueError("GridSpec requires either `values` or (min_, max_, n)")
91:         if self.n <= 0:
92:             raise ValueError("GridSpec.n must be positive")
93:         if self.scale.lower() in ("lin", "linear", "linspace"):
94:             arr = np.linspace(self.min_, self.max_, int(self.n), dtype=self.dtype)
95:         elif self.scale.lower() in ("log10", "log", "logspace"):
96:             # interpret bounds as exponents (base 10)
97:             exps = np.linspace(self.min_, self.max_, int(self.n), dtype=self.dtype)
98:             arr = np.power(10.0, exps, dtype=self.dtype)
99:         else:
100:             raise ValueError(f"Unknown scale {self.scale!r}")
101:         return arr
102: 
103: 
104: def _parse_param_spec(obj: object, dtype=np.float64) -> GridSpec:
105:     """Parse a parameter spec from a value/list/dict."""
106:     if isinstance(obj, dict):
107:         # keys could be min/max/n/scale
108:         min_ = obj.get("min", obj.get("min_", None))
109:         max_ = obj.get("max", obj.get("max_", None))
110:         n = obj.get("n", None)
111:         scale = obj.get("scale", "lin")
112:         return GridSpec(values=None, min_=min_, max_=max_, n=n, scale=scale, dtype=dtype)
113:     # list/array/scalar → explicit values
114:     vals = np.asarray(obj, dtype=dtype).reshape(-1)
115:     return GridSpec(values=vals, dtype=dtype)
116:

119:     """Turn a {name: spec} dict into {name: 1D array}."""

130: def named_product(grid_dict: Mapping[str, Array]) -> List[Dict[str, float]]:
131:     """List of dicts for each grid point."""
132:     keys = list(grid_dict.keys())
133:     vals = [np.asarray(grid_dict[k]).reshape(-1) for k in keys]
134:     combos = []
135:     for tup in product(*vals):
136:         combos.append({k: float(v) for k, v in zip(keys, tup)})
137:     return combos

141:     """Stacked 2D array of shape (N, P) with parameter order."""
142:     if order is None:
143:         order = list(grid_dict.keys())
144:     cols = [np.asarray(grid_dict[k]).reshape(-1) for k in order]
145:     N = int(np.prod([c.size for c in cols], dtype=int))
146:     P = len(cols)

148:     # fill by iterating product
149:     idx = 0

160:     meshes = np.meshgrid(*arrays, indexing="ij")

===== palatini_pt/numerics/solvers.py =====
1: # palatini_pt/numerics/solvers.py
2: # -*- coding: utf-8 -*-
3: """
4: Small linear algebra / spectral helpers with NumPy-only core.
5: 
6: Highlights
7: ----------
8: - Eigendecomposition (symmetric) & spectral utilities
9: - Generalized symmetric eigenproblem A v = λ B v  (B ≻ 0) via Cholesky
10: - Stable Cholesky with jitter fallback
11: - Nullspace (SVD), pseudo-inverse, stable least squares
12: - PSD checks and projections (for DoF/degeneracy tests later)
13: """
14: from __future__ import annotations
15: 
16: from dataclasses import dataclass
17: from typing import Iterable, Optional, Tuple
18: 
19: import numpy as np
20: 
21: Array = np.ndarray
22: ArrayLike = np.ndarray | Iterable[float]
23: 
24: 
25: # -----------------------------
26: # Basic helpers
27: # -----------------------------
28: 
29: def symmetrize(A: Array) -> Array:
30:     A = np.asarray(A, dtype=float)
31:     return 0.5 * (A + A.T)
32: 
33: 
34: def antisymmetrize(A: Array) -> Array:
35:     A = np.asarray(A, dtype=float)
36:     return 0.5 * (A - A.T)
37: 
38: 
39: def spectral_radius(A: Array) -> float:
40:     """ρ(A) = max |λ_i| (not assuming symmetry)."""
41:     w = np.linalg.eigvals(np.asarray(A, dtype=float))
42:     return float(np.max(np.abs(w)))
43: 
44: 
45: # -----------------------------
46: # Eigen (symmetric)
47: # -----------------------------
48: 
49: def eigvals_symmetric(A: Array, *, sort: str = "asc") -> Array:
50:     """Eigenvalues of a real symmetric matrix."""
51:     w = np.linalg.eigvalsh(symmetrize(A))
52:     if sort == "asc":
53:         return w
54:     if sort == "desc":
55:         return w[::-1]
56:     return w
57: 
58: 
59: def eigh_symmetric(A: Array, *, sort: str = "asc") -> Tuple[Array, Array]:
60:     """(w, V) for symmetric A."""
61:     w, V = np.linalg.eigh(symmetrize(A))
62:     if sort == "asc":
63:         return w, V
64:     if sort == "desc":
65:         idx = np.argsort(w)[::-1]
66:         return w[idx], V[:, idx]
67:     return w, V
68: 
69: 
70: # -----------------------------
71: # Cholesky (with jitter)
72: # -----------------------------
73: 
74: @dataclass
75: class CholeskyResult:
76:     L: Array
77:     jitter: float
78:     attempts: int
79: 
80: 
81: def safe_cholesky(A: Array, *, jitter0: float = 1e-12, factor: float = 10.0, max_tries: int = 8) -> CholeskyResult:
82:     """Try LL^T with increasing jitter*I until success or give up."""
83:     A = symmetrize(A)
84:     n = A.shape[0]
85:     I = np.eye(n, dtype=float)
86:     jitter = 0.0
87:     for t in range(max_tries + 1):
88:         try:
89:             L = np.linalg.cholesky(A + jitter * I)
90:             return CholeskyResult(L=L, jitter=jitter, attempts=t + 1)
91:         except np.linalg.LinAlgError:
92:             if t == 0:
93:                 jitter = jitter0
94:             else:
95:                 jitter *= factor
96:     # last attempt: eigen-based floor
97:     w, V = np.linalg.eigh(A)
98:     w = np.maximum(w, 0.0)
99:     Afloor = (V * w) @ V.T
100:     L = np.linalg.cholesky(Afloor + jitter * I)
101:     return CholeskyResult(L=L, jitter=jitter, attempts=max_tries + 2)
102: 
103: 
104: def is_posdef(A: Array, *, tol: float = 0.0) -> bool:
105:     """Check positive-definiteness (optionally allow tiny negatives via tol)."""
106:     try:
107:         cr = safe_cholesky(A, jitter0=max(tol, 1e-16))
108:         return True
109:     except Exception:
110:         w = eigvals_symmetric(A)
111:         return bool(np.min(w) > -abs(tol))
112: 
113: 
114: def is_semidefinite(A: Array, *, tol: float = 1e-12) -> bool:
115:     """Check positive-semidefinite within tolerance."""
116:     w = eigvals_symmetric(A)
117:     return bool(np.min(w) >= -abs(tol))
118: 
119: 
120: def psd_projection(A: Array, *, floor: float = 0.0) -> Array:
121:     """Project symmetric A to PSD by zeroing negative eigenvalues (≥ floor)."""
122:     w, V = np.linalg.eigh(symmetrize(A))
123:     w = np.maximum(w, floor)
124:     return (V * w) @ V.T
125: 
126: 
127: # -----------------------------
128: # Generalized symmetric eigenproblem
129: # -----------------------------
130: 
131: def generalized_eigh(A: Array, B: Array) -> Tuple[Array, Array]:
132:     """Solve A v = λ B v assuming B ≻ 0; returns (w, V) with B-orthonormal V."""
133:     A = symmetrize(A)
134:     B = symmetrize(B)
135:     # Cholesky of B
136:     chol = safe_cholesky(B)
137:     L = chol.L
138:     # transform: A' = L^{-1} A L^{-T}, then eigh
139:     Linv = np.linalg.inv(L)
140:     A_tilde = Linv @ A @ Linv.T
141:     w, Q = np.linalg.eigh(A_tilde)
142:     # back-transform eigenvectors: V = L^{-T} Q
143:     V = np.linalg.solve(L.T, Q)
144:     # Normalize so that V^T B V = I
145:     # (should already hold numerically; enforce softly)
146:     for i in range(V.shape[1]):
147:         norm = np.sqrt(V[:, i].T @ B @ V[:, i])
148:         if norm > 0:
149:             V[:, i] /= norm
150:     return w, V
151: 
152: 
153: # -----------------------------
154: # Linear solves & SVD tools
155: # -----------------------------
156: 
157: def stable_solve(A: Array, b: Array, *, rcond: float = 1e-12) -> Tuple[Array, float]:
158:     """Least-squares solver with residual norm; handles rank-deficient."""
159:     A = np.asarray(A, dtype=float)
160:     b = np.asarray(b, dtype=float)
161:     x, residuals, rank, s = np.linalg.lstsq(A, b, rcond=rcond)
162:     res = float(np.sqrt(residuals.sum())) if residuals.size else float(np.linalg.norm(A @ x - b))
163:     return x, res
164: 
165: 
166: def pinv(A: Array, *, rcond: float = 1e-12) -> Array:
167:     """Moore-Penrose pseudo-inverse."""
168:     return np.linalg.pinv(np.asarray(A, dtype=float), rcond=rcond)
169: 
170: 
171: def nullspace(A: Array, *, rtol: float = 1e-12) -> Array:
172:     """Right-nullspace basis as columns (possibly empty with shape (n,0))."""
173:     A = np.asarray(A, dtype=float)
174:     u, s, vh = np.linalg.svd(A, full_matrices=True)
175:     tol = rtol * (s[0] if s.size else 1.0)
176:     mask = s <= tol
177:     # When full_matrices=True, vh.shape = (n, n). Nullspace vectors correspond to trailing rows.
178:     # But numerical mask is more robust:
179:     r = np.sum(~mask)
180:     ns = vh[r:, :].T  # (n, n-r)
181:     return ns
182: 
183: 
184: def cond_number(A: Array) -> float:
185:     """2-norm condition number."""
186:     u, s, vh = np.linalg.svd(np.asarray(A, dtype=float), full_matrices=False)
187:     if s.size == 0:
188:         return 0.0
189:     return float((s[0] / s[-1]) if s[-1] > 0 else np.inf)
190: 
191: 
192: def blockdiag(*mats: Array) -> Array:
193:     """Block diagonal concatenate."""
194:     mats = [np.asarray(M, dtype=float) for M in mats]
195:     m = sum(M.shape[0] for M in mats)
196:     n = sum(M.shape[1] for M in mats)
197:     out = np.zeros((m, n), dtype=float)
198:     i = j = 0
199:     for M in mats:
200:         h, w = M.shape
201:         out[i : i + h, j : j + w] = M
202:         i += h
203:         j += w
204:     return out
205:

===== palatini_pt/numerics/validate.py =====
1: # palatini_pt/numerics/validate.py
2: # -*- coding: utf-8 -*-
3: """
4: Validation utilities: tolerances, sanity checks, and small diagnostics.
5: 
6: Use cases (Phase 1+)
7: --------------------
8: - Near-zero tests with dual thresholds (sym_zero / num_zero)
9: - Symmetry checks for matrices
10: - Relative error helpers for reports
11: - Limit probing (e → 0) using a user callback
12: - Monotonic convergence checks (e.g., IBP threshold scans later)
13: """
14: from __future__ import annotations
15: 
16: from dataclasses import dataclass
17: from typing import Callable, Iterable, Tuple
18: 
19: import numpy as np
20: 
21: Array = np.ndarray
22: 
23: 
24: # -----------------------------
25: # Tolerances
26: # -----------------------------
27: 
28: @dataclass
29: class Tolerances:
30:     sym_zero: float = 1.0e-12  # symbolic-level "zero"
31:     num_zero: float = 1.0e-9   # numerical-level "zero"
32:     eig_floor: float = 1.0e-12
33:     rel: float = 1.0e-8
34: 
35: 
36: # -----------------------------
37: # Basic predicates
38: # -----------------------------
39: 
40: def near_zero(x, *, atol: float) -> bool:
41:     """True if |x| (or sup-norm of array) <= atol."""
42:     arr = np.asarray(x, dtype=float)
43:     return bool(np.max(np.abs(arr)) <= abs(atol))
44: 
45: 
46: def zero_mask(x, *, atol: float) -> Array:
47:     """Elementwise |x| <= atol as bool array."""
48:     arr = np.asarray(x, dtype=float)
49:     return np.abs(arr) <= abs(atol)
50: 
51: 
52: def assert_near_zero(x, *, atol: float, name: str = "value") -> None:
53:     if not near_zero(x, atol=atol):
54:         arr = np.asarray(x, dtype=float)
55:         raise AssertionError(f"{name} not near zero: max|.|={np.max(np.abs(arr)):.3e} > {atol:.3e}")
56: 
57: 
58: def assert_symmetric(A: Array, *, atol: float = 1e-12, name: str = "matrix") -> None:
59:     A = np.asarray(A, dtype=float)
60:     dev = np.max(np.abs(A - A.T))
61:     if dev > atol:
62:         raise AssertionError(f"{name} not symmetric: max|A-A^T|={dev:.3e} > {atol:.3e}")
63: 
64: 
65: def rel_error(a, b, *, mode: str = "max") -> float:
66:     """Relative error with different denominators.
67: 
68:     mode:
69:       - "max": |a-b| / max(1, |a|, |b|)
70:       - "a"  : |a-b| / max(1, |a|)
71:       - "b"  : |a-b| / max(1, |b|)
72:       - "mean": |a-b| / max(1, 0.5*(|a|+|b|))
73:     """
74:     a = np.asarray(a, dtype=float)
75:     b = np.asarray(b, dtype=float)
76:     num = np.max(np.abs(a - b))
77:     if mode == "max":
78:         den = max(1.0, float(np.max(np.abs(a))), float(np.max(np.abs(b))))
79:     elif mode == "a":
80:         den = max(1.0, float(np.max(np.abs(a))))
81:     elif mode == "b":
82:         den = max(1.0, float(np.max(np.abs(b))))
83:     elif mode == "mean":
84:         den = max(1.0, float(0.5 * (np.max(np.abs(a)) + np.max(np.abs(b)))))
85:     else:
86:         raise ValueError("Unknown mode for rel_error.")
87:     return float(num / den)
88: 
89: 
90: def assert_allclose(a, b, *, atol: float, rtol: float = 0.0, name: str = "value") -> None:
91:     a = np.asarray(a, dtype=float)
92:     b = np.asarray(b, dtype=float)
93:     if not np.allclose(a, b, atol=atol, rtol=rtol):
94:         diff = np.max(np.abs(a - b))
95:         raise AssertionError(f"{name} not close: max|Δ|={diff:.3e} > atol={atol:.3e} (rtol={rtol:.1e})")
96: 
97: 
98: # -----------------------------
99: # Convergence & limits
100: # -----------------------------
101: 
102: def is_monotone(xs: Iterable[float], *, kind: str = "nonincreasing", atol: float = 0.0) -> bool:
103:     """Check monotonic trend up to tolerance."""
104:     arr = np.asarray(list(xs), dtype=float).reshape(-1)
105:     if arr.size < 2:
106:         return True
107:     dif = np.diff(arr)
108:     if kind == "nonincreasing":
109:         return bool(np.all(dif <= atol))
110:     if kind == "nondecreasing":
111:         return bool(np.all(dif >= -atol))
112:     raise ValueError("kind must be 'nonincreasing' or 'nondecreasing'")
113: 
114: 
115: def probe_limit_zero(
116:     f: Callable[[float], float],
117:     epsilons: Iterable[float],
118:     *,
119:     atol: float = 1e-9,
120:     require_monotone: bool = False,
121: ) -> Tuple[float, bool, bool]:
122:     """Evaluate f(ε) as ε→0⁺ using a decreasing sequence.
123: 
124:     Returns
125:     -------
126:     (last_value, below_atol, monotone_ok)
127:     """
128:     eps = np.asarray(list(epsilons), dtype=float).reshape(-1)
129:     if eps.size == 0:
130:         raise ValueError("epsilons must be non-empty")
131:     ys = np.array([float(f(e)) for e in eps], dtype=float)
132:     last = float(ys[-1])
133:     below = bool(np.abs(last) <= atol)
134:     mono_ok = True
135:     if require_monotone:
136:         mono_ok = is_monotone(np.abs(ys), kind="nonincreasing", atol=atol * 0.1)
137:     return last, below, mono_ok
138: 
139: 
140: def rms(arr) -> float:
141:     arr = np.asarray(arr, dtype=float)
142:     return float(np.sqrt(np.mean(arr * arr))) if arr.size else 0.0

===== palatini_pt/palatini/connection.py =====
1: # palatini_pt/palatini/connection.py
2: # -*- coding: utf-8 -*-
3: """
4: Connection and contorsion helpers (minimal, algebraic).
5: 
6: 目標
7: ----
8: 本模組提供在「純跡扭率」假設下，連接與扭率/擬曲率的最小建構：
9: - 以 metric g_{μν} 與「扭率跡向量」T_μ 生成對應的扭率張量
10:   T_{λμν} = (1/3)(g_{λν} T_μ - g_{λμ} T_ν)
11: - 對應的 contorsion（純跡情形）
12:   K_{λμν} = (1/3)(g_{λμ} T_ν - g_{λν} T_μ)
13: 
14: 這些式子對應到經典分解的純跡分量；當你需要升級到
15: 一般連接 Γ = { } + K 時，只要把 Levi-Civita({}) 接成數值或符號即可。
16: """
17: from __future__ import annotations
18: 
19: from typing import Optional
20: 
21: import numpy as np
22: 
23: 
24: def minkowski_metric(sign: int = -1) -> np.ndarray:
25:     """回傳 4D Minkowski metric（diag(sign, +1, +1, +1)）。
26: 
27:     Parameters
28:     ----------
29:     sign : int
30:         時間分量的符號，預設 -1（mostly plus）。
31:     """
32:     return np.diag([sign, 1.0, 1.0, 1.0])
33: 
34: 
35: def torsion_from_trace(T_vec: np.ndarray, g: Optional[np.ndarray] = None) -> np.ndarray:
36:     """由「扭率跡向量」建立純跡扭率張量 T_{λμν}。
37: 
38:     定義
39:     ----
40:     T_{λμν} = (1/3)(g_{λν} T_μ - g_{λμ} T_ν)
41: 
42:     Notes
43:     -----
44:     - 回傳的 T 具有反稱性 T_{λμν} = - T_{λνμ}
45:     - 維度預設 4；若 g 提供其他維度亦可，但公式仍採上述純跡型式。
46:     """
47:     if g is None:
48:         g = minkowski_metric()
49:     g = np.asarray(g, dtype=float)
50:     T_vec = np.asarray(T_vec, dtype=float)
51:     dim = g.shape[0]
52:     assert T_vec.shape == (dim,)
53: 
54:     T = np.zeros((dim, dim, dim), dtype=float)
55:     for lam in range(dim):
56:         for mu in range(dim):
57:             for nu in range(dim):
58:                 T[lam, mu, nu] = (1.0 / 3.0) * (g[lam, nu] * T_vec[mu] - g[lam, mu] * T_vec[nu])
59:     return T
60: 
61: 
62: def contorsion_from_trace(T_vec: np.ndarray, g: Optional[np.ndarray] = None) -> np.ndarray:
63:     """純跡扭率對應的 contorsion K_{λμν}。
64: 
65:     K_{λμν} = (1/3)(g_{λμ} T_ν - g_{λν} T_μ)
66:     """
67:     if g is None:
68:         g = minkowski_metric()
69:     g = np.asarray(g, dtype=float)
70:     T_vec = np.asarray(T_vec, dtype=float)
71:     dim = g.shape[0]
72:     assert T_vec.shape == (dim,)
73: 
74:     K = np.zeros((dim, dim, dim), dtype=float)
75:     for lam in range(dim):
76:         for mu in range(dim):
77:             for nu in range(dim):
78:                 K[lam, mu, nu] = (1.0 / 3.0) * (g[lam, mu] * T_vec[nu] - g[lam, nu] * T_vec[mu])
79:     return K
80:

===== palatini_pt/palatini/field_eq.py =====
1: # palatini_pt/palatini/field_eq.py
2: # -*- coding: utf-8 -*-
3: """
4: Algebraic field equations (C1): torsion is pure trace and aligns with ∂ε.
5: 
6: 最小模型
7: --------
8: 假設扭率沒有動力學，僅由 spurion 的梯度（外源）代數決定：

11:

13: ----

15: - 純跡扭率張量 T_{λμν}（由 T_μ 生成）

17: """

19: 
20: from dataclasses import dataclass

22:

===== palatini_pt/palatini/torsion_decomp.py =====
1: # palatini_pt/palatini/torsion_decomp.py
2: # -*- coding: utf-8 -*-
3: """
4: Torsion decomposition: trace / axial / pure-tensor pieces.
5: 
6: 分解式（4D，全降指）：
7:     T_{λμν} = (1/3)(g_{λν} T_μ - g_{λμ} T_ν) - (1/6) ε_{λμνσ} S^σ + q_{λμν}
8: 其中
9:     T_μ      = T^{λ}{}_{μλ} = g^{λρ} T_{ρ μ λ}          （跡向量）
10:     S^σ      = ε^{σμνρ} T_{μνρ}                         （軸向向量）
11:     q_{λμν}  = 剩餘純張量（trace-free & axial-free）
12: 
13: 提供：
14: - `trace_vector(T, g, ginv)`
15: - `axial_vector(T, ginv)`
16: - `decompose(T, g)` → dict 包含 trace/axial/pure_tensor 與重建檢查
17: """
18: from __future__ import annotations
19: 
20: from typing import Dict, Tuple
21: 
22: import numpy as np
23: 
24: 
25: def levi_civita_4() -> np.ndarray:
26:     """回傳 4D Levi-Civita ε_{αβγδ}，採座標正向定義（0123 = +1）。"""
27:     eps = np.zeros((4, 4, 4, 4), dtype=int)
28:     import itertools
29: 
30:     for perm in itertools.permutations(range(4)):
31:         sign = permutation_sign(perm)
32:         eps[perm] = sign
33:     return eps
34: 
35: 
36: def permutation_sign(perm: Tuple[int, int, int, int]) -> int:
37:     """簽位函式：偶排列 +1、奇排列 -1。"""
38:     sign = 1
39:     p = list(perm)
40:     for i in range(len(p)):
41:         for j in range(i + 1, len(p)):
42:             if p[i] > p[j]:
43:                 sign *= -1
44:     return sign
45: 
46: 
47: def minkowski_metric(sign: int = -1) -> np.ndarray:
48:     return np.diag([sign, 1.0, 1.0, 1.0])
49: 
50: 
51: def inverse_metric(g: np.ndarray) -> np.ndarray:
52:     return np.linalg.inv(g)
53: 
54: 
55: def trace_vector(T: np.ndarray, ginv: np.ndarray) -> np.ndarray:
56:     """T_μ = T^{λ}{}_{μλ} = g^{λρ} T_{ρ μ λ}（全降指輸入）。"""
57:     dim = T.shape[0]
58:     out = np.zeros((dim,), dtype=float)
59:     for mu in range(dim):
60:         s = 0.0
61:         for lam_up in range(dim):
62:             for rho in range(dim):
63:                 s += ginv[lam_up, rho] * T[rho, mu, lam_up]
64:         out[mu] = s
65:     return out
66: 
67: 
68: def axial_vector(T: np.ndarray, ginv: np.ndarray) -> np.ndarray:
69:     """S^σ = ε^{σμνρ} T_{μνρ}（全降指輸入，藉由 g^{-1} 把 ε 升指）。"""
70:     dim = T.shape[0]
71:     assert dim == 4, "目前軸向向量實作限定 4D"
72:     eps_dn = levi_civita_4().astype(float)  # 下標版本
73:     # 以 g^{-1} 升第一個指標：ε^{σ}{}_{μνρ} = g^{σα} ε_{α μ ν ρ}
74:     out = np.zeros((dim,), dtype=float)
75:     for sig in range(dim):
76:         s = 0.0
77:         for alpha in range(dim):
78:             for mu in range(dim):
79:                 for nu in range(dim):
80:                     for rho in range(dim):
81:                         eps_up = 0.0
82:                         # ε^{σ μ ν ρ} = g^{σα} g^{μβ} g^{νγ} g^{ρδ} ε_{αβγδ}，但只需要與 T_{μνρ}
83:                         # 收縮第一個指標即可簡化成：
84:                         eps_up += ginv[sig, alpha] * eps_dn[alpha, mu, nu, rho]
85:                         s += eps_up * T[mu, nu, rho]
86:         out[sig] = s
87:     return out
88: 
89: 
90: def pure_trace_piece(T_vec: np.ndarray, g: np.ndarray) -> np.ndarray:
91:     """(1/3)(g_{λν} T_μ - g_{λμ} T_ν)"""
92:     dim = g.shape[0]
93:     P = np.zeros((dim, dim, dim), dtype=float)
94:     for lam in range(dim):
95:         for mu in range(dim):
96:             for nu in range(dim):
97:                 P[lam, mu, nu] = (1.0 / 3.0) * (g[lam, nu] * T_vec[mu] - g[lam, mu] * T_vec[nu])
98:     return P
99: 
100: 
101: def axial_piece(S_vec: np.ndarray) -> np.ndarray:
102:     """-(1/6) ε_{λμνσ} S^σ，限定 4D。"""
103:     dim = S_vec.shape[0]
104:     assert dim == 4, "目前 axial piece 實作限定 4D"
105:     eps = levi_civita_4().astype(float)
106:     A = np.zeros((dim, dim, dim), dtype=float)
107:     for lam in range(dim):
108:         for mu in range(dim):
109:             for nu in range(dim):
110:                 s = 0.0
111:                 for sig in range(dim):
112:                     s += eps[lam, mu, nu, sig] * S_vec[sig]
113:                 A[lam, mu, nu] = - (1.0 / 6.0) * s
114:     return A
115: 
116: 
117: def decompose(T: np.ndarray, g: np.ndarray | None = None) -> Dict[str, np.ndarray]:
118:     """做三分解：trace/axial/pure_tensor，並回傳重建檢查殘差。"""
119:     if g is None:
120:         g = minkowski_metric()
121:     ginv = inverse_metric(g)
122: 
123:     T_tr = trace_vector(T, ginv)
124:     S_ax = axial_vector(T, ginv)
125:     P = pure_trace_piece(T_tr, g)
126:     A = axial_piece(S_ax)
127:     q = T - P - A
128:     recon = P + A + q
129:     res = T - recon
130:     return {
131:         "trace_vec": T_tr,
132:         "axial_vec": S_ax,
133:         "pure_trace": P,
134:         "axial": A,
135:         "pure_tensor": q,
136:         "reconstruct_residual": res,
137:     }
138: 
139: 
140: __all__ = [
141:     "minkowski_metric",
142:     "inverse_metric",
143:     "trace_vector",
144:     "axial_vector",
145:     "pure_trace_piece",
146:     "axial_piece",
147:     "decompose",
148: ]

===== palatini_pt/spurion/profile.py =====
1: # palatini_pt/spurion/profile.py
2: # -*- coding: utf-8 -*-
3: """
4: Spurion profiles ε(x): value, gradient, and box (d'Alembert/Laplacian).
5: 
6: Phase 1 features
7: ----------------
8: - ConstantProfile:          ε = const
9: - PlaneWaveProfile:         ε = A cos(k·x + φ); grad = -A sin(...) k; box = - (k^T g k) A cos(...)
10: - GaussianRadialProfile:    ε = A exp(-(r/R)^2), r^2 = x^T x
11: - CustomProfile:            user-defined callables (with finite-diff fallback)
12: 
13: All profiles expose:
14:     - dim               (int)
15:     - metric            (np.ndarray)    # g^{μν} contravariant, can be diagonal or full
16:     - epsilon(x)        -> float
17:     - grad(x)           -> np.ndarray shape (dim,)
18:     - box(x)            -> float        # g^{μν} ∂_μ ∂_ν ε
19: 
20: Notes
21: -----
22: - Default metric is Euclidean identity of size `dim`.
23: - For `PlaneWaveProfile`, `box` uses the supplied `metric` exactly: -(k^T g k) * A cos(...).
24: - For `GaussianRadialProfile`, `box` uses general diagonal or full metric via analytic Hessian.
25: - For `CustomProfile`, if grad/box are not provided, robust central finite differences are used,
26:   including cross-derivatives for a full-metric contraction.
27: 
28: This module is dependency-light (NumPy only) and ready for tests in Phase 1.
29: """
30: from __future__ import annotations
31: 
32: from dataclasses import dataclass
33: from typing import Callable, Optional, Sequence, Tuple, Union
34: 
35: import numpy as np
36: 
37: Array = np.ndarray
38: ArrayLike = Union[Sequence[float], np.ndarray]
39: 
40: 
41: # -----------------------------
42: # Utilities
43: # -----------------------------
44: 
45: 
46: def _as_1d(x: ArrayLike) -> Array:
47:     x = np.asarray(x, dtype=float).reshape(-1)
48:     if x.ndim != 1:
49:         raise ValueError("x must be a 1D vector.")
50:     return x
51: 
52: 
53: def _as_metric(g: Optional[ArrayLike], dim: int) -> Array:
54:     """Accept None | scalar | 1D diag | 2D full and return a (dim, dim) array."""
55:     if g is None:
56:         return np.eye(dim, dtype=float)
57:     G = np.asarray(g, dtype=float)
58:     if G.ndim == 0:
59:         return float(G) * np.eye(dim, dtype=float)
60:     if G.ndim == 1:
61:         if G.shape[0] != dim:
62:             raise ValueError(f"diag metric length {G.shape[0]} != dim {dim}")
63:         return np.diag(G)
64:     if G.ndim == 2:
65:         if G.shape != (dim, dim):
66:             raise ValueError(f"metric shape {G.shape} incompatible with dim {dim}")
67:         return G
68:     raise ValueError("metric must be None, scalar, 1D diag, or 2D square.")
69: 
70: 
71: def _fd_grad(f: Callable[[Array], float], x: Array, h: float = 1e-6) -> Array:
72:     """Central finite-diff gradient."""
73:     n = x.size
74:     g = np.zeros(n, dtype=float)
75:     for i in range(n):
76:         e = np.zeros(n); e[i] = 1.0
77:         fp = f(x + h * e)
78:         fm = f(x - h * e)
79:         g[i] = (fp - fm) / (2.0 * h)
80:     return g
81: 
82: 
83: def _fd_hessian(f: Callable[[Array], float], x: Array, h: float = 1e-5) -> Array:
84:     """Central finite-diff Hessian (symmetric)."""
85:     n = x.size
86:     H = np.zeros((n, n), dtype=float)
87:     f0 = f(x)
88:     # Diagonal second derivatives
89:     for i in range(n):
90:         e_i = np.zeros(n); e_i[i] = 1.0
91:         fpp = f(x + h * e_i)
92:         fmm = f(x - h * e_i)
93:         H[i, i] = (fpp - 2.0 * f0 + fmm) / (h ** 2)
94:     # Mixed second derivatives
95:     for i in range(n):
96:         e_i = np.zeros(n); e_i[i] = 1.0
97:         for j in range(i + 1, n):
98:             e_j = np.zeros(n); e_j[j] = 1.0
99:             fpp = f(x + h * e_i + h * e_j)
100:             fpm = f(x + h * e_i - h * e_j)
101:             fmp = f(x - h * e_i + h * e_j)
102:             fmm = f(x - h * e_i - h * e_j)
103:             val = (fpp - fpm - fmp + fmm) / (4.0 * h * h)
104:             H[i, j] = H[j, i] = val
105:     return H
106: 
107: 
108: # -----------------------------
109: # Base interface (duck-typed)
110: # -----------------------------
111: 
112: 
113: @dataclass
114: class BaseProfile:
115:     dim: int
116:     metric: Optional[ArrayLike] = None  # g^{μν}
117: 
118:     def __post_init__(self):
119:         if self.dim <= 0:
120:             raise ValueError("dim must be positive.")
121:         self._G = _as_metric(self.metric, self.dim)  # (dim, dim)
122: 
123:     # API to implement
124:     def epsilon(self, x: ArrayLike) -> float:  # pragma: no cover - abstract
125:         raise NotImplementedError
126: 
127:     def grad(self, x: ArrayLike) -> Array:  # pragma: no cover - abstract
128:         raise NotImplementedError
129: 
130:     def box(self, x: ArrayLike) -> float:  # pragma: no cover - abstract
131:         raise NotImplementedError
132: 
133:     # Convenience
134:     @property
135:     def G(self) -> Array:
136:         """Contravariant metric g^{μν} as a full (dim,dim) array."""
137:         return self._G
138: 
139: 
140: # -----------------------------
141: # Profiles
142: # -----------------------------
143: 
144: 
145: @dataclass
146: class ConstantProfile(BaseProfile):
147:     value: float = 1.0
148: 
149:     def epsilon(self, x: ArrayLike) -> float:
150:         _ = _as_1d(x)  # validate length if desired
151:         return float(self.value)
152: 
153:     def grad(self, x: ArrayLike) -> Array:
154:         x = _as_1d(x)
155:         return np.zeros(self.dim, dtype=float)
156: 
157:     def box(self, x: ArrayLike) -> float:
158:         _ = _as_1d(x)
159:         return 0.0
160: 
161: 
162: @dataclass
163: class PlaneWaveProfile(BaseProfile):
164:     amplitude: float = 1.0
165:     kvec: ArrayLike = None  # shape (dim,)
166:     phase: float = 0.0
167:     use_cosine: bool = True
168: 
169:     def __post_init__(self):
170:         super().__post_init__()
171:         if self.kvec is None:
172:             raise ValueError("PlaneWaveProfile requires kvec.")
173:         self._k = _as_1d(self.kvec)
174:         if self._k.size != self.dim:
175:             raise ValueError(f"kvec length {self._k.size} != dim {self.dim}")
176: 
177:     def _phase(self, x: Array) -> float:
178:         return float(np.dot(self._k, x) + self.phase)
179: 
180:     def epsilon(self, x: ArrayLike) -> float:
181:         x = _as_1d(x)
182:         arg = self._phase(x)
183:         if self.use_cosine:
184:             return float(self.amplitude * np.cos(arg))
185:         else:
186:             return float(self.amplitude * np.sin(arg))
187: 
188:     def grad(self, x: ArrayLike) -> Array:
189:         x = _as_1d(x)
190:         arg = self._phase(x)
191:         if self.use_cosine:
192:             # d/dx cos = -sin * k
193:             return -self.amplitude * np.sin(arg) * self._k
194:         else:
195:             # d/dx sin =  cos * k
196:             return self.amplitude * np.cos(arg) * self._k
197: 
198:     def box(self, x: ArrayLike) -> float:
199:         x = _as_1d(x)
200:         arg = self._phase(x)
201:         # ∂μ∂ν ε = -A cos(arg) kμ kν  (for cosine), or -A sin(arg) kμ kν (for sine)
202:         if self.use_cosine:
203:             h_scalar = -self.amplitude * np.cos(arg)
204:         else:
205:             h_scalar = -self.amplitude * np.sin(arg)
206:         # g^{μν} kμ kν
207:         quad = float(self._k @ self.G @ self._k)
208:         return float(h_scalar * quad)
209: 
210: 
211: @dataclass
212: class GaussianRadialProfile(BaseProfile):
213:     amplitude: float = 1.0
214:     radius: float = 1.0  # R in exp(-(r/R)^2)
215: 
216:     def _r2(self, x: Array) -> float:
217:         return float(np.dot(x, x))
218: 
219:     def epsilon(self, x: ArrayLike) -> float:
220:         x = _as_1d(x)
221:         R2 = self.radius ** 2
222:         return float(self.amplitude * np.exp(-self._r2(x) / R2))
223: 
224:     def grad(self, x: ArrayLike) -> Array:
225:         x = _as_1d(x)
226:         eps = self.epsilon(x)
227:         return (-2.0 / (self.radius ** 2)) * eps * x
228: 
229:     def hessian(self, x: ArrayLike) -> Array:
230:         """Analytic Hessian for ε = A exp(-(x·x)/R^2)."""
231:         x = _as_1d(x)
232:         n = x.size
233:         eps = self.epsilon(x)
234:         R2 = self.radius ** 2
235:         # ∂i∂j ε = ε * [ (-2/R^2) δij + (4/(R^4)) x_i x_j ]
236:         H = (-2.0 / R2) * eps * np.eye(n) + (4.0 / (R2 ** 2)) * eps * np.outer(x, x)
237:         return H
238: 
239:     def box(self, x: ArrayLike) -> float:
240:         x = _as_1d(x)
241:         H = self.hessian(x)
242:         return float(np.einsum("ij,ij->", self.G, H))  # g^{ij} H_{ij}
243: 
244: 
245: @dataclass
246: class CustomProfile(BaseProfile):
247:     f_epsilon: Callable[[Array], float] = None
248:     f_grad: Optional[Callable[[Array], Array]] = None
249:     f_box: Optional[Callable[[Array], float]] = None
250:     fd_step_grad: float = 1e-6
251:     fd_step_hess: float = 1e-5
252: 
253:     def __post_init__(self):
254:         super().__post_init__()
255:         if self.f_epsilon is None:
256:             raise ValueError("CustomProfile requires `f_epsilon` callable.")
257: 
258:     def epsilon(self, x: ArrayLike) -> float:
259:         x = _as_1d(x)
260:         return float(self.f_epsilon(x))
261: 
262:     def grad(self, x: ArrayLike) -> Array:
263:         x = _as_1d(x)
264:         if self.f_grad is not None:
265:             g = np.asarray(self.f_grad(x), dtype=float).reshape(-1)
266:             if g.size != self.dim:
267:                 raise ValueError("Custom grad size mismatch with dim.")
268:             return g
269:         # finite-diff
270:         return _fd_grad(self.f_epsilon, x, h=self.fd_step_grad)
271: 
272:     def box(self, x: ArrayLike) -> float:
273:         x = _as_1d(x)
274:         if self.f_box is not None:
275:             return float(self.f_box(x))
276:         # Use full Hessian (incl. cross terms) then contract with metric
277:         H = _fd_hessian(self.f_epsilon, x, h=self.fd_step_hess)
278:         return float(np.einsum("ij,ij->", self.G, H))
279:

288:     dim: int,
289:     metric: Optional[ArrayLike] = None,
290:     **kwargs,
291: ) -> BaseProfile:
292:     """
293:     Factory for common spurion profiles.
294: 
295:     Parameters
296:     ----------
297:     kind : {"constant","plane","gaussian","custom"}
298:     dim  : int
299:     metric : array-like | None
300:         Contravariant metric g^{μν}. None -> I.
301: 
302:     Returns
303:     -------
304:     BaseProfile

306:     k = kind.lower()
307:     if k in ("constant", "const"):

320:     """
321:     Return a constant ∂ε vector (as a 1D numpy array).
322:     If normalize=True, return vec / ||vec||.

326:         n = float(np.linalg.norm(v))
327:         if n == 0.0:
328:             raise ValueError("cannot normalize zero vector")

337:     If metric is None, use identity.
338:     """
339:     v = _as_1d(vec)
340:     G = _as_metric(metric, v.size)

===== palatini_pt/spurion/pt_even.py =====
1: # palatini_pt/spurion/pt_even.py
2: # -*- coding: utf-8 -*-
3: """
4: PT-even selector for monomials built from basic building blocks.
5: 
6: Design goals (Phase 1)
7: ----------------------
8: - Minimal, explicit parity bookkeeping (±1) for named atoms.
9: - Parse simple multiplicative expressions with optional exponents:
10:     "grad_eps^2 * eps", "box * eps", "d * d * eps", "g * grad_eps * grad_eps"
11: - Provide:
12:     - PTRegistry: registry of atom → parity (±1), extensible.
13:     - compute_parity(expr): +1 or -1 (strict by default).
14:     - pt_project(exprs): keep only PT-even.
15:     - assert_pt_even(exprs): assert all are PT-even, else raise with details.
16: 
17: Conventions (defaults; can be changed via registry):
18: ----------------------------------------------------
19: - "eps" / "epsilon"        : +1   (spurion scalar)
20: - "d", "partial", "nabla"  : -1   (derivative operator under PT)
21: - "grad_eps"               : -1   (one derivative acting on eps)
22: - "box", "lap_eps"         : +1   (two derivatives on eps contracted)
23: - "g"                      : +1   (metric tensor)
24: You can register extra atoms (e.g., "torsion_T") as needed.
25: 
26: These rules are self-consistent for typical O(∂^2) invariants:
27: - (grad_eps)^2  → (-1)^2 = +1  (PT-even)
28: - box * eps     → (+1)*(+1) = +1
29: - d * eps       → -1            (PT-odd, will be projected out)
30: 
31: This module is algebra-agnostic: it does not depend on SymPy or the rest
32: of the pipeline; it just classifies & filters strings.
33: """
34: from __future__ import annotations
35: 
36: import re
37: import numpy as np
38: from dataclasses import dataclass, field
39: from typing import Dict, Iterable, List, Sequence, Tuple, Any
40: 
41: 
42: 
43: 
44: # -----------------------------
45: # Registry
46: # -----------------------------
47: 
48: 
49: @dataclass
50: class PTRegistry:
51:     """
52:     Mapping from atom name to PT parity (+1 or -1).
53: 
54:     Parameters
55:     ----------
56:     mapping : dict[str, int] | None
57:         Seed mapping; if None, defaults are used.
58: 
59:     Methods
60:     -------
61:     register(name: str, parity: int)
62:     parity(name: str) -> int
63:     """
64:     mapping: Dict[str, int] = field(default_factory=dict)
65: 
66:     def __post_init__(self):
67:         if not self.mapping:
68:             self.mapping.update(
69:                 {
70:                     "eps": +1,
71:                     "epsilon": +1,
72:                     "d": -1,
73:                     "partial": -1,
74:                     "nabla": -1,
75:                     "grad_eps": -1,
76:                     "lap_eps": +1,
77:                     "box": +1,
78:                     "g": +1,
79:                 }
80:             )
81: 
82:     def register(self, name: str, parity: int) -> None:
83:         if parity not in (+1, -1):
84:             raise ValueError("parity must be ±1.")
85:         self.mapping[name] = int(parity)
86: 
87:     def parity(self, name: str) -> int:
88:         if name not in self.mapping:
89:             raise KeyError(f"Unknown atom '{name}'. Register it or pass strict=False.")
90:         return int(self.mapping[name])
91: 
92: 
93: # -----------------------------
94: # Expression parsing
95: # -----------------------------
96: 
97: _TOKEN = r"[A-Za-z_][A-Za-z0-9_\-]*"
98: _TERM_RE = re.compile(rf"\s*({_TOKEN})(?:\s*\^\s*([+-]?\d+))?\s*")
99: 
100: 
101: def _parse_expr(expr: str) -> List[Tuple[str, int]]:
102:     """
103:     Parse a simple product expression into [(name, power), ...].
104: 
105:     Grammar (multiplicative only):
106:         EXPR := TERM ( ('*'|whitespace) TERM )*
107:         TERM := NAME [ '^' INT ]
108: 
109:     Examples
110:     --------
111:     "grad_eps^2 * eps"    -> [("grad_eps", 2), ("eps", 1)]
112:     "d * d * eps"         -> [("d", 1), ("d", 1), ("eps", 1)]
113:     "box*epsilon"         -> [("box", 1), ("epsilon", 1)]
114:     """
115:     s = expr.strip()
116:     if not s:
117:         raise ValueError("Empty expression.")
118:     # split by '*' OR whitespace; we re-scan term by term
119:     parts = [p for p in re.split(r"\*", s) if p.strip()]
120:     tokens: List[Tuple[str, int]] = []
121:     for part in parts:
122:         # a part may still contain whitespace-separated terms; tokenize greedily
123:         pos = 0
124:         text = part.strip()
125:         while pos < len(text):
126:             m = _TERM_RE.match(text, pos)
127:             if not m:
128:                 # try to consume a single whitespace and continue
129:                 if text[pos].isspace():
130:                     pos += 1
131:                     continue
132:                 raise ValueError(f"Cannot parse around: {text[pos:pos+16]!r}")
133:             name = m.group(1)
134:             pow_str = m.group(2)
135:             power = int(pow_str) if pow_str is not None else 1
136:             tokens.append((name, power))
137:             pos = m.end()
138:     return tokens
139: 
140: 
141: # -----------------------------
142: # Parity evaluation
143: # -----------------------------
144: 
145: 
146: def compute_parity(
147:     expr: str | Sequence[Tuple[str, int]],
148:     registry: PTRegistry | None = None,
149:     *,
150:     strict: bool = True,
151:     unknown_default: int = +1,
152: ) -> int:
153:     """
154:     Compute PT parity of an expression.
155: 
156:     Parameters
157:     ----------
158:     expr : str | list[(name, power)]
159:         Multiplicative expression (see _parse_expr).
160:     registry : PTRegistry | None
161:         Parity registry (uses defaults if None).
162:     strict : bool
163:         If True, unknown atom raises. If False, use `unknown_default`.
164:     unknown_default : int
165:         Used when strict=False and atom not in registry.
166: 
167:     Returns
168:     -------
169:     +1 or -1
170:     """
171:     reg = registry or PTRegistry()
172:     terms = _parse_expr(expr) if isinstance(expr, str) else list(expr)
173:     parity = +1
174:     for name, power in terms:
175:         if strict:
176:             p = reg.parity(name)
177:         else:
178:             p = reg.mapping.get(name, unknown_default)
179:             if p not in (+1, -1):
180:                 p = unknown_default
181:         # parity multiplies; negative power flips accordingly but we reduce mod 2
182:         if power % 2 != 0:
183:             parity *= p
184:     return +1 if parity >= 0 else -1
185:

187: def is_pt_even(expr: str | Sequence[Tuple[str, int]], registry: PTRegistry | None = None) -> bool:
188:     return compute_parity(expr, registry=registry) == +1

191: def pt_project(
192:     exprs: Iterable[str | Sequence[Tuple[str, int]]],
193:     registry: PTRegistry | None = None,
194: ) -> List[str]:
195:     """
196:     Keep only PT-even expressions. Returns them as strings (round-tripped if needed).
197:     """
198:     reg = registry or PTRegistry()
199:     kept: List[str] = []
200:     for e in exprs:
201:         s = e if isinstance(e, str) else " * ".join(f"{n}^{p}" if p != 1 else n for n, p in e)
202:         if is_pt_even(e, registry=reg):
203:             kept.append(s)
204:     return kept

207: def assert_pt_even(
208:     exprs: Iterable[str | Sequence[Tuple[str, int]]],
209:     registry: PTRegistry | None = None,
210: ) -> None:
211:     """
212:     Assert all expressions are PT-even. Raises AssertionError listing offenders.
213:     """
214:     reg = registry or PTRegistry()
215:     odd: List[str] = []
216:     for e in exprs:
217:         s = e if isinstance(e, str) else " * ".join(f"{n}^{p}" if p != 1 else n for n, p in e)
218:         if not is_pt_even(e, registry=reg):
219:             odd.append(s)
220:     if odd:
221:         raise AssertionError("Found PT-odd monomials (project them out): " + ", ".join(odd))
