
===== notebooks/00_sanity.py =====
1: # %% [markdown]
2: # # 00 — sanity
3: # Quick import + TT luminality smoke (should be exactly 1 at LO).
4: 
5: # %%
6: import numpy as np
7: from palatini_pt.gw.quadratic_action import tensor_action_coeffs
8: from palatini_pt.spurion.pt_even import project_scalar
9: from palatini_pt.spurion.profile import constant_gradient
10: 
11: # %%
12: k = np.logspace(-3, -1, 64)
13: deps = constant_gradient([0.0, 0.0, 1.0])  # ∂ε 指向 z
14: Seps = project_scalar(np.dot(deps, deps))
15: 
16: K, G = tensor_action_coeffs(Seps)
17: cT2 = G / K
18: print("K(LO)=", K, "G(LO)=", G, "c_T^2(LO)=", cT2)
19: assert np.allclose(cT2, 1.0), "locked LO must be luminal"

===== notebooks/10_c2_symbolic_demo.py =====
1: # %% [markdown]
2: # # 10 — C2 symbolic demo
3: # Show DBI / closed-metric reduce to A_* * I_T at quadratic order.
4: 
5: # %%
6: import sympy as sp
7: from palatini_pt.equivalence.coeff_extractor import reduce_to_IT
8: 
9: lam, eta, Seps = sp.symbols("lambda eta Seps", positive=True)
10: res_dbi = reduce_to_IT(route="dbi", lam=lam, eta=eta, Seps=Seps)
11: res_cm  = reduce_to_IT(route="closed_metric", lam=lam, eta=eta, Seps=Seps)
12: sp.simplify(res_dbi - res_cm)

===== notebooks/30_nlo_offsets.py =====
1: # %% [markdown]
2: # # 30 — NLO offsets
3: # δc_T^2(k) ~ b k^2 / Λ^2
4: 
5: # %%
6: import numpy as np, matplotlib.pyplot as plt
7: from palatini_pt.gw.nlo import predict_offsets
8: 
9: k = np.logspace(-3, -1, 200)
10: cfg = {"nlo": {"gradT_sq_eff": 1.0, "Ricci_deps_deps_eff": 0.3, "Lambda2": 1e6}}
11: out = predict_offsets(k, cfg)
12: 
13: plt.loglog(k, np.abs(out["delta_cT2"]), label=r"$|\delta c_T^2|$")
14: plt.loglog(k, (out["delta_cT2"][0]/(k[0]**2))*k**2, ls="--", label=r"$\propto k^2$")
15: plt.legend(); plt.xlabel("k"); plt.ylabel(r"$|\delta c_T^2|$")
16: plt.title("NLO: $k^2/\\Lambda^2$")
17: plt.show()
