
===== scripts/fig_c1_alignment.py =====
1: #!/usr/bin/env python3
2: # -*- coding: utf-8 -*-
3: """
4: Fig.2 — C1: alignment angle between T_mu and ∂_mu ε.
5: Outputs:
6:   figs/pdf/fig2_c1_alignment.pdf
7:   figs/data/c1_alignment.csv (+ .md5)
8: """
9: from __future__ import annotations
10: 
11: import argparse
12: import csv
13: import hashlib
14: import os
15: from pathlib import Path
16: from typing import Dict, List
17: 
18: import matplotlib.pyplot as plt
19: import numpy as np
20: 
21: REQUIRE_REAL = bool(int(os.environ.get("PALPT_REQUIRE_REAL_APIS", "0")))
22: 
23: 
24: def _apply_style():
25:     try:
26:         from palatini_pt.plotting.style import apply_prd_style  # type: ignore
27:         apply_prd_style()
28:     except Exception:
29:         pass
30: 
31: 
32: def _prepare_outdirs(config: Dict | None) -> Dict[str, Path]:
33:     base = Path((config or {}).get("output", {}).get("dir", "figs"))
34:     pdf = base / "pdf"
35:     data = base / "data"
36:     png = base / "png"
37:     for d in (pdf, data, png):
38:         d.mkdir(parents=True, exist_ok=True)
39:     return {"pdf": pdf, "data": data, "png": png}
40: 
41: 
42: def _write_md5(path: Path) -> Path:
43:     h = hashlib.md5()
44:     with open(path, "rb") as f:
45:         for chunk in iter(lambda: f.read(8192), b""):
46:             h.update(chunk)
47:     m = path.with_suffix(path.suffix + ".md5")
48:     m.write_text(h.hexdigest() + "\n", encoding="utf-8")
49:     return m
50: 
51: 
52: def _alignment_via_module(config: Dict | None, n: int) -> np.ndarray:
53:     import importlib
54:     mod = importlib.import_module("palatini_pt.palatini.c1_pure_trace")
55:     for name in ["alignment_samples", "sample_alignment_angles", "alignment_scan"]:
56:         if hasattr(mod, name):
57:             fn = getattr(mod, name)
58:             arr = fn(config=config, n=n)
59:             return np.asarray(arr, dtype=float).ravel()
60:     for name in ["alignment_angle", "angle_alignment"]:
61:         if hasattr(mod, name):
62:             ang = float(getattr(mod, name)(config=config))
63:             return np.array([ang], dtype=float)
64:     raise RuntimeError("No alignment API found in c1_pure_trace")
65: 
66: 
67: def _alignment_fallback(n: int, seed: int = 0) -> np.ndarray:
68:     rng = np.random.default_rng(seed)
69:     return np.abs(rng.normal(loc=0.0, scale=1e-3, size=n))
70: 
71: 
72: def run(config: Dict | None = None, which: str = "full") -> Dict[str, List[str]]:
73:     _apply_style()
74:     paths = _prepare_outdirs(config)
75:     n = 200 if which != "smoke" else 40
76: 
77:     try:
78:         angles = _alignment_via_module(config, n)
79:     except Exception:
80:         if REQUIRE_REAL:
81:             raise
82:         angles = _alignment_fallback(n)
83: 
84:     # write CSV
85:     csv_path = paths["data"] / "c1_alignment.csv"
86:     with open(csv_path, "w", newline="", encoding="utf-8") as f:
87:         w = csv.writer(f)
88:         w.writerow(["idx", "angle_rad"])
89:         for i, a in enumerate(angles):
90:             w.writerow([i, f"{float(a):.16e}"])
91:     _write_md5(csv_path)
92: 
93:     # histogram
94:     fig, ax = plt.subplots(figsize=(4.2, 3.0))
95:     ax.hist(angles, bins=30 if n > 60 else 12, density=True)
96:     ax.set_xlabel("alignment angle (rad)")
97:     ax.set_ylabel("PDF")
98:     ax.set_title("C1: alignment of $T_\\mu$ and $\\partial_\\mu\\epsilon$")
99:     # ---- 容差線 & 註記統計 ----
100:     ang_tol = float((config or {}).get("c1", {}).get("angle_tol", 1e-3))
101:     ax.axvline(ang_tol, ls="--", lw=1.0, color="C1", label=f"tol={ang_tol:g}")
102:     mean = float(np.mean(angles))
103:     p95  = float(np.percentile(angles, 95))
104:     ax.text(
105:         0.98, 0.92,
106:         f"mean={mean:.2e}\n95%={p95:.2e}",
107:         transform=ax.transAxes, ha="right", va="top", fontsize=9
108:     )
109:     if ang_tol is not None:
110:         ax.legend(frameon=False, fontsize=9)
111:     fig.tight_layout()
112:     pdf_path = paths["pdf"] / "fig2_c1_alignment.pdf"
113:     fig.savefig(pdf_path, bbox_inches="tight")
114:     plt.close(fig)
115:     _write_md5(pdf_path)
116: 
117:     return {"pdfs": [str(pdf_path)], "data": [str(csv_path)]}
118: 
119: 
120: def _load_config_if_needed(path: str | None) -> Dict | None:
121:     if not path:
122:         return None
123:     try:
124:         from palatini_pt.io.config import load_config  # type: ignore
125:         return load_config(path)
126:     except Exception:
127:         import yaml  # type: ignore
128:         with open(path, "r", encoding="utf-8") as f:
129:             return yaml.safe_load(f)
130: 
131: 
132: def main():
133:     ap = argparse.ArgumentParser(description="Fig.2 — C1 alignment")
134:     ap.add_argument("--config", default="configs/default.yaml")
135:     args = ap.parse_args()
136:     out = run(_load_config_if_needed(args.config))
137:     print(out)
138: 
139: 
140: if __name__ == "__main__":
141:     main()

52: def _compute_components_via_module(config: Dict | None) -> Tuple[float, float, float]:
53:     """
54:     Try multiple API names from palatini_pt.palatini.c1_pure_trace.
55:     Return (trace, axial, tensor) norms.
56:     """
57:     import importlib
58:     c1 = importlib.import_module("palatini_pt.palatini.c1_pure_trace")
59:     for name in ["compute_components", "scan_components", "pure_trace_components"]:
60:         if hasattr(c1, name):
61:             fn = getattr(c1, name)
62:             out = fn(config=config)
63:             if isinstance(out, dict):
64:                 t = float(out.get("trace", 0.0))
65:                 a = float(out.get("axial", 0.0))
66:                 s = float(out.get("tensor", 0.0))
67:                 return t, a, s
68:             if isinstance(out, (list, tuple)) and len(out) >= 3:
69:                 return float(out[0]), float(out[1]), float(out[2])
70:     try:
71:         fe = importlib.import_module("palatini_pt.palatini.field_eq")
72:         if hasattr(fe, "torsion_components"):
73:             t, a, s = fe.torsion_components(config=config)
74:             return float(t), float(a), float(s)
75:     except Exception:
76:         pass
77:     raise RuntimeError("Could not find a suitable API to compute C1 components.")

80: def _compute_components_fallback(seed: int = 42) -> Tuple[float, float, float]:
81:     rng = np.random.default_rng(seed)
82:     trace = 1.0 + 0.05 * rng.normal()

84:     tensor = 1e-6 * abs(rng.normal())

===== scripts/fig_c2_coeff_compare.py =====
1: # scripts/fig_c2_coeff_compare.py
2: # -*- coding: utf-8 -*-
3: from __future__ import annotations
4: 
5: import csv
6: import os
7: from pathlib import Path
8: from typing import Dict, List
9: 
10: import matplotlib.pyplot as plt
11: import numpy as np
12: 
13: REQUIRE_REAL = bool(int(os.environ.get("PALPT_REQUIRE_REAL_APIS", "0")))
14: 
15: 
16: def _apply_style():
17:     try:
18:         from palatini_pt.plotting.style import apply_prd
19:         apply_prd()
20:     except Exception:
21:         pass
22: 
23: 
24: def _prepare_outdirs(config: Dict | None) -> Dict[str, Path]:
25:     outdir = Path(config.get("output", {}).get("dir", "figs")) if config else Path("figs")
26:     pdfdir = outdir / "pdf"
27:     datadir = outdir / "data"
28:     pdfdir.mkdir(parents=True, exist_ok=True)
29:     datadir.mkdir(parents=True, exist_ok=True)
30:     return {"pdfdir": pdfdir, "datadir": datadir}
31:

33: def _write_md5(p: Path):
34:     import hashlib
35:     h = hashlib.md5(p.read_bytes()).hexdigest()
36:     (p.with_suffix(p.suffix + ".md5")).write_text(h + "\n")
37: 
38: 
39: def _residual_scan_via_module(config: Dict | None, thresholds: np.ndarray) -> np.ndarray:
40:     import importlib
41:     # order2_checker API variants
42:     names = [
43:         ("palatini_pt.equivalence.order2_checker", "residual_scan"),
44:         ("palatini_pt.equivalence.order2_checker", "scan_residuals"),
45:     ]
46:     for modname, fnname in names:
47:         try:
48:             mod = importlib.import_module(modname)
49:             if hasattr(mod, fnname):
50:                 fn = getattr(mod, fnname)
51:                 arr = fn(config=config, thresholds=thresholds)
52:                 return np.asarray(arr, dtype=float).ravel()
53:         except Exception:
54:             continue
55:     # coeff_extractor direct residual
56:     try:
57:         mod = importlib.import_module("palatini_pt.equivalence.coeff_extractor")
58:         for fnname in ["residual_norm", "compare_residual_norm"]:
59:             if hasattr(mod, fnname):
60:                 fn = getattr(mod, fnname)
61:                 vals = [float(fn(config=config, ibp_tol=float(th))) for th in thresholds]
62:                 return np.array(vals, dtype=float)
63:     except Exception:
64:         pass
65:     raise RuntimeError("No residual-scan API found for equivalence/")
66: 
67: 
68: def _residual_scan_fallback(thresholds: np.ndarray) -> np.ndarray:
69:     floor = 1e-12
70:     start = 1e-4
71:     span = max(1e-12, float(np.ptp(np.asarray(thresholds))))
72:     vals = start * np.exp(-3.0 * (thresholds - float(np.min(thresholds))) / span)
73:     return np.maximum(floor, vals)
74: 
75: 
76: def run(config: Dict | None = None, which: str = "full") -> Dict[str, List[str]]:
77:     _apply_style()
78:     paths = _prepare_outdirs(config)
79: 
80:     thresholds = np.logspace(-14, -8, 13) if which != "smoke" else np.logspace(-12, -10, 5)
81:     try:
82:         resids = _residual_scan_via_module(config, thresholds)
83:     except Exception:
84:         if REQUIRE_REAL:
85:             raise
86:         resids = _residual_scan_fallback(thresholds)
87: 
88:     # --- 畫圖 ---
89:     fig, ax = plt.subplots(figsize=(4.5, 3.2))
90:     ax.semilogx(thresholds, resids, marker="o", lw=1.2)
91:     ax.set_xlabel("IBP tolerance")
92:     ax.set_ylabel(r"$\|\Delta\|$")
93:     ax.set_title("C2 residual (smoke)" if which == "smoke" else "C2 residual scan")
94:     ax.grid(True, ls=":", alpha=0.6)
95:     
96:     # ---- 目標殘差水平線 + 達成標註 ----
97:     resid_goal = float((config or {}).get("equivalence", {}).get("resid_goal", 1e-10))
98:     ax.axhline(resid_goal, ls="--", lw=1.0, color="C1", label=f"goal={resid_goal:.1e}")
99:     try:
100:         # 找到第一個低於目標的 IBP 容差
101:         mask = resids <= resid_goal
102:         if np.any(mask):
103:             th_hit = thresholds[np.argmax(mask)]
104:             ax.axvline(th_hit, ls=":", lw=1.0, color="C2")
105:             ax.text(
106:                 th_hit, resid_goal, f"  hit @ {th_hit:.1e}",
107:                 va="bottom", ha="left", fontsize=8, color="C2"
108:             )
109:         ax.legend(frameon=False, fontsize=9)
110:     except Exception:
111:         pass
112: 
113:     pdf = paths["pdfdir"] / ("fig3_c2_coeff_compare.pdf" if which != "smoke" else "fig3_c2_coeff_compare_smoke.pdf")
114:     fig.tight_layout()
115:     fig.savefig(pdf, dpi=200)
116:     plt.close(fig)
117:     _write_md5(pdf)
118: 
119:     # --- 資料 ---
120:     data_path = paths["datadir"] / ("c2_residuals.csv" if which != "smoke" else "c2_residuals_smoke.csv")
121:     with data_path.open("w", newline="") as f:
122:         w = csv.writer(f)
123:         w.writerow(["ibp_tol", "residual_norm"])
124:         for t, r in zip(thresholds, resids):
125:             w.writerow([f"{t:.16e}", f"{r:.16e}"])
126:     _write_md5(data_path)
127: 
128:     return {"pdfs": [str(pdf)], "data": [str(data_path)]}
129: 
130: 
131: if __name__ == "__main__":
132:     print(run(which="smoke"))

52: def _compute_grid_via_module(config: Dict | None) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
53:     # 先嘗試直接的 cT_grid()
54:     import importlib
55:     for modname, fnname in [
56:         ("palatini_pt.gw.tensor_mode", "cT_grid"),
57:         ("palatini_pt.gw.tensor_mode", "grid_cT"),
58:         ("palatini_pt.gw.tensor_mode", "compute_cT_grid"),
59:     ]:
60:         try:
61:             mod = importlib.import_module(modname)
62:             if hasattr(mod, fnname):
63:                 fn = getattr(mod, fnname)
64:                 g = fn(config=config)
65:                 if isinstance(g, dict):
66:                     X = np.asarray(g["X"])
67:                     Y = np.asarray(g["Y"])
68:                     cT = np.asarray(g["cT"])
69:                 else:
70:                     X, Y, cT = g
71:                     X, Y, cT = map(np.asarray, (X, Y, cT))
72:                 return X, Y, cT
73:         except Exception:
74:             continue
75: 
76:     # 沒有 cT_grid：用真實 API cT_of_k 掃格（param1 = seps_scale 的縮放、param2 = k）
77:     try:
78:         mod = importlib.import_module("palatini_pt.gw.tensor_mode")
79:         if not hasattr(mod, "cT_of_k"):
80:             raise RuntimeError("gw.tensor_mode.cT_of_k not found")
81: 
82:         cT_of_k = getattr(mod, "cT_of_k")
83: 
84:         grids = (config or {}).get("grids", {}).get("ct", {})
85:         n_s = int(grids.get("n_s", 61))
86:         n_k = int(grids.get("n", grids.get("nk", 61)))
87:         kmin = float(grids.get("kmin", 1e-4))
88:         kmax = float(grids.get("kmax", 1e-1))
89: 
90:         sval = np.linspace(0.0, 1.0, n_s)  # seps_scale 的比例 0..1
91:         kval = np.logspace(np.log10(kmin), np.log10(kmax), n_k)
92: 
93:         X, Y = np.meshgrid(sval, kval, indexing="xy")  # X: s, Y: k
94:         cT = np.empty_like(X, dtype=float)
95: 
96:         base_seps = float((config or {}).get("spurion", {}).get("seps_scale", 1.0))
97: 
98:         for j in range(n_k):
99:             cfg_j = deepcopy(config) if config else {}
100:             spur = cfg_j.setdefault("spurion", {})
101:             spur["seps_scale"] = base_seps  # 先放底值
102:             # 對每個 s 調整 seps_scale 再算 cT(k)
103:             for i in range(n_s):
104:                 spur["seps_scale"] = base_seps * float(sval[i])
105:                 cT[j, i] = float(cT_of_k(k=np.array([kval[j]]), config=cfg_j, locked=False)[0])
106: 
107:         return X, Y, cT
108: 
109:     except Exception as e:
110:         raise RuntimeError(f"Cannot build cT heatmap via module APIs: {e}")
111: 
112:

===== scripts/mixing_matrix_extract.py =====
1: #!/usr/bin/env python3
2: """
3: Paper artifact: extract (or stage) the 2x2 mixing coefficients used in App. D.
4: 
5: Usage (JSON → CSV + meta + .md5):
6:   python -m scripts.mixing_matrix_extract \
7:       --from-json configs/coeffs/mixing_matrix_FRW.json \
8:       --out-csv   figs/data/mixing_matrix.csv \
9:       --out-meta  figs/data/mixing_matrix_meta.json \
10:       --lock-tol  1e-6 \
11:       --require-lock
12: 
13: JSON schema (minimal):
14: {
15:   "N":    { "ROD": <float>, "CM": <float> },
16:   "divN": { "ROD": <float>, "CM": <float> }
17: }
18: 
19: This tool:
20:   1) Writes a normalized CSV: entry,mu_ROD,mu_CM  (entries: N, divN)
21:   2) Verifies the locking ratio consistency:
22:        rN   = - mu_CM(N)    / mu_ROD(N)
23:        rDiv = - mu_CM(divN) / mu_ROD(divN)
24:      and checks |rN - rDiv| <= tol
25:   3) Emits a meta JSON with the ratios and w* (locked) up to normalization
26:   4) Writes .md5 fingerprints for CSV and meta (for reproducibility)
27:   5) Exits with nonzero code if --require-lock and the ratios mismatch
28: 
29: Notes:
30:   * If any mu_ROD == 0, locking ratio is ill-defined → error.
31:   * This script is intentionally lightweight; physics-side extraction
32:     should produce the JSON. Here we only stage, validate, and log.
33: """
34: 
35: from __future__ import annotations
36: import argparse, json, os, sys, hashlib, time
37: from typing import Dict, Any, Tuple
38: 
39: try:
40:     import pandas as pd  # part of dev deps
41: except Exception:
42:     sys.stderr.write("Error: pandas not installed. Run 'make install'.\n")
43:     raise
44: 
45: 
46: def _write_md5(path: str) -> None:
47:     h = hashlib.md5()
48:     with open(path, "rb") as f:
49:         for chunk in iter(lambda: f.read(8192), b""):
50:             h.update(chunk)
51:     with open(path + ".md5", "w") as g:
52:         g.write(h.hexdigest() + "\n")
53: 
54: 
55: def _need_pair(d: Dict[str, Any], key: str) -> Dict[str, float]:
56:     if key not in d:
57:         raise KeyError(f"Missing key '{key}' in JSON.")
58:     sub = d[key]
59:     if not all(k in sub for k in ("ROD", "CM")):
60:         raise KeyError(f"Entry '{key}' must contain 'ROD' and 'CM'.")
61:     return {"ROD": float(sub["ROD"]), "CM": float(sub["CM"])}
62: 
63: 
64: def load_mu_from_json(fp: str) -> Dict[str, Dict[str, float]]:
65:     with open(fp, "r") as f:
66:         obj: Dict[str, Any] = json.load(f)
67:     mu_N    = _need_pair(obj, "N")
68:     mu_divN = _need_pair(obj, "divN")
69:     return {"N": mu_N, "divN": mu_divN}
70: 
71: 
72: def locking_ratios(mu: Dict[str, Dict[str, float]]) -> Tuple[float, float]:
73:     """Return (rN, rDiv) where r = - mu_CM / mu_ROD for each entry."""
74:     def r(entry: str) -> float:
75:         rod = mu[entry]["ROD"]
76:         cm  = mu[entry]["CM"]
77:         if abs(rod) == 0.0:
78:             raise ZeroDivisionError(f"mu_ROD({entry}) == 0 makes locking ratio ill-defined.")
79:         return - cm / rod
80:     return r("N"), r("divN")
81: 
82: 
83: def write_outputs(mu: Dict[str, Dict[str, float]],
84:                   out_csv: str,
85:                   out_meta: str,
86:                   tol: float,
87:                   require_lock: bool,
88:                   print_summary: bool = True) -> int:
89:     """Write CSV + meta (+md5). Return process exit code (0 ok / 1 fail)."""
90:     # ensure out dirs
91:     os.makedirs(os.path.dirname(os.path.abspath(out_csv)) or ".", exist_ok=True)
92:     os.makedirs(os.path.dirname(os.path.abspath(out_meta)) or ".", exist_ok=True)
93: 
94:     # CSV
95:     df = pd.DataFrame(
96:         [
97:             {"entry": "N",    "mu_ROD": mu["N"]["ROD"],    "mu_CM": mu["N"]["CM"]},
98:             {"entry": "divN", "mu_ROD": mu["divN"]["ROD"], "mu_CM": mu["divN"]["CM"]},
99:         ]
100:     )
101:     df.to_csv(out_csv, index=False)
102:     _write_md5(out_csv)
103: 
104:     # Ratios & lock check
105:     rN, rDiv = locking_ratios(mu)
106:     lock_ok = abs(rN - rDiv) <= tol
107: 
108:     # A canonical locked weight up to normalization (choose w_CM*=1)
109:     w_star = {
110:         "w_ROD_star_over_w_CM_star": rN,  # equals rDiv if lock_ok
111:         "w_ROD_star": rN,
112:         "w_CM_star": 1.0,
113:     }
114: 
115:     # Meta JSON
116:     meta = {
117:         "timestamp_unix": time.time(),
118:         "inputs": {
119:             "mu": mu,
120:         },
121:         "outputs": {
122:             "csv": out_csv,
123:             "csv_md5": out_csv + ".md5",
124:             "meta": out_meta,
125:         },
126:         "locking_check": {
127:             "ratio_N": rN,
128:             "ratio_divN": rDiv,
129:             "abs_diff": abs(rN - rDiv),
130:             "tol": tol,
131:             "lock_ok": lock_ok,
132:         },
133:         "locked_weights": w_star,
134:         "notes": (
135:             "Locking requires the two ratios to match within tol. "
